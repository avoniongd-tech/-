const DRIVE_FOLDER_ID = 'REPLACE_WITH_FOLDER_ID';
const TARGET_SUBPATH = 'lf';
function deploy(){
  const root=DriveApp.getFolderById(DRIVE_FOLDER_ID);
  const site=ensurePath_(root,TARGET_SUBPATH);
  const pub=ensurePath_(site,'public');
  const assets=ensurePath_(pub,'assets');
  const css=ensurePath_(assets,'css');
  const js=ensurePath_(assets,'js');
  const img=ensurePath_(assets,'img');
  const icons=ensurePath_(assets,'icons');
  const video=ensurePath_(assets,'video');
  const data=ensurePath_(site,'data');
  const admin=ensurePath_(site,'admin');
  writeText_(pub,'preloader.html','<!doctype html><title>Preloader</title>');
  writeText_(pub,'index.html','<!doctype html><title>Index</title>');
  writeText_(css,'style.css','/* styles */');
  writeText_(js,'app.js','// js');
  writeText_(data,'content.json','{}');
  writeText_(admin,'login.html','<!doctype html><title>Login</title>');
  writeText_(admin,'panel.html','<!doctype html><title>Panel</title>');
  Logger.log('Ready: '+site.getUrl());
}
function ensurePath_(parent,path){
  return path.split('/').reduce((cur,seg)=>{seg=seg.trim();if(!seg)return cur;const it=cur.getFoldersByName(seg);return it.hasNext()?it.next():cur.createFolder(seg);}, parent);
}
function writeText_(folder,name,content){
  const it=folder.getFilesByName(name);while(it.hasNext())it.next().setTrashed(true);
  return folder.createFile(name, content, MimeType.HTML);
}
